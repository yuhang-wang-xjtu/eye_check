cmake_minimum_required(VERSION 3.20)
project(EyeBreak202020 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(EyeBreak WIN32 src/main.cpp src/reminder.rc)
target_include_directories(EyeBreak PRIVATE assets)
target_compile_definitions(EyeBreak PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(EyeBreak PRIVATE /W4 /utf-8)
target_compile_options(EyeBreak PRIVATE /Zi)
target_link_options(EyeBreak PRIVATE /DEBUG /PDBALTPATH:EyeBreak.pdb)
target_link_options(EyeBreak PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
target_link_libraries(EyeBreak PRIVATE user32 gdi32 shell32 winmm comdlg32 xinput hid wtsapi32 dwmapi comctl32 uxtheme)
enable_testing()
add_executable(timer_tests tests/timer_tests.cpp)
target_compile_options(timer_tests PRIVATE /W4 /utf-8)
